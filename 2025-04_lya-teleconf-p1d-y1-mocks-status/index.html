\mathcal{C}_{1\mathrm{D}, \alpha}(A, B, z) = \left(\frac{\sum_{s \in z} w'_{A,s} \sum_{s \in z} w'_{B,s}}{\sum_{s \in z} w'_{A,s} w'_{B,s}} -1\right)^{-1} \left[ \frac{\sum_{s \in z} w'_{A,s} w'_{B,s} P_{A,s} P_{B,s}}{\sum_{s \in z} w'_{A,s} w'_{B,s}} - \left\langle P_{A,s} \right\rangle_{s \in z} \left\langle P_{B,s} \right\rangle_{s \in z}\right]\,.
\sigma_{1\mathrm{D},\alpha}^2 (A,z) = \left(\frac{(\sum_{s \in z} w'_{A,s})^2 }{\sum_{s \in z} (w'_{A,s})^2} -1\right)^{-1} \left[ \frac{\sum_{s \in z} (w'_{A,s})^2 P_{A,s}^2}{\sum_{s \in z} (w'_{A,s})^2} - \left\langle P_{A,s} \right\rangle_{s \in z}^2 \right]\,,
P_{A,s} = \sum_{k \in A_s} p_s(k) / N_{A,s}
w_{A,s}^\prime = N_{A,s} w_{A,s}
